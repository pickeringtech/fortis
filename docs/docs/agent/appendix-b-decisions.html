Correction patterns#

or die → try main (LR 4.6)#

Fortis
// accept
import fortis.host { Console }

export error Empty {
  code: "demo.empty"
}

try func first(items: ListView[Int]) Int {
  if items.count() == 0 {
    fail Empty()
  }
  return items.get(0) ?? 0
}

try func main(console: Console) {
  let n = first(List.of(4, 5)) or fail
  console.print("${n}")
}
output
4
